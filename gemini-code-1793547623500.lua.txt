local network

-- GUI Creation Helper
local function Create(class, parent, props)
	local new = Instance.new(class)
	for k, v in next, props do
		new[k] = v
	end
	new.Parent = parent
	return new
end

-- Character setup
local character, root, humanoid
local function onCharacterAdded(char)
	if not char then return end
	character = char
	root = character:WaitForChild("HumanoidRootPart", 5)
	humanoid = character:WaitForChild("Humanoid", 5)
	if humanoid then
		pcall(function()
			humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
		end)
	end
end
onCharacterAdded(player.Character or player.CharacterAdded:Wait())
player.CharacterAdded:Connect(onCharacterAdded)

local function instantTP(cf)
	if not root then return end
	pcall(function()
		root.CFrame = cf
		root.Velocity = Vector3.new()
		root.RotVelocity = Vector3.new()
		if humanoid then
			humanoid.Sit = false
		end
	end)
end

if not isTargetGame or not network then
	statusLbl.Text = "Status: Target game elements missing"
	return
end

-- Supply tracking setup
local supplyCounts = {TomatoSauce = 99, Cheese = 99, Sausage = 99, Pepperoni = 99, Dough = 99, Box = 99, Dew = 99}
pcall(function()
	for name in pairs(supplyCounts) do
		local counterObj = workspace:FindFirstChild("SupplyCounters")
		if counterObj then
			local lbl = counterObj.Model[name == "Dew" and "CounterMountainDew" or "Counter" .. name].a.SG.Counter
			supplyCounts[name] = tonumber(lbl.Text) or 99
			lbl.Changed:Connect(function()
				supplyCounts[name] = tonumber(lbl.Text) or 99
			end)
		end
	end
end)

local workspaceOrders = workspace:FindFirstChild("Orders")
local workspaceAllMountainDew = workspace:FindFirstChild("AllMountainDew")
local workspaceAllDough = workspace:FindFirstChild("AllDough")
local workspaceOvens = workspace:FindFirstChild("Ovens")
local workspaceSupplyButtons = workspace:FindFirstChild("SupplyButtons")
local workspaceAllSupplyBoxes = workspace:FindFirstChild("AllSupplyBoxes")

local orderDict = {["3540529228"] = "Cheese", ["3540530535"] = "Sausage", ["3540529917"] = "Pepperoni", ["2512571151"] = "Dew", ["2512441325"] = "Dew"}
local cookingDict = {Cheese = 0, Sausage = 0, Pepperoni = 0, Dew = 0}
local cookPtick = 0
local cookDtick = 0

local function getOrders()
	local orders = {}
	if not workspaceOrders then return orders end
	local tempCookingDict = {Cheese = cookingDict.Cheese, Sausage = cookingDict.Sausage, Pepperoni = cookingDict.Pepperoni, Dew = cookingDict.Dew}
	local children = workspaceOrders:GetChildren()
	for i = 1, #children do
		local sg = children[i]:FindFirstChild("SG")
		if sg then
			local imageLabel = sg:FindFirstChild("ImageLabel")
			if imageLabel then
				local o = orderDict[imageLabel.Image:match("%d+$")]
				if o then
					if tempCookingDict[o] > 0 then
						tempCookingDict[o] = tempCookingDict[o] - 1
					elseif (o == "Dew" and workspaceAllMountainDew and #workspaceAllMountainDew:GetChildren() > 0) or (o ~= "Dew" and supplyCounts[o] > 0 and supplyCounts.TomatoSauce > 0 and supplyCounts.Cheese > 0) then
						orders[#orders + 1] = o
					end
				end
			end
		end
	end
	return orders
end

local function FindFirstDew()
	if not workspaceAllMountainDew then return end
	for _, d in ipairs(workspaceAllMountainDew:GetChildren()) do
		local isBurned = d:FindFirstChild("IsBurned")
		if (isBurned == nil or isBurned.Value == false) and not d.Anchored then
			return d
		end
	end
end

local function FindBadDew()
	if not workspaceAllMountainDew then return end
	for _, d in ipairs(workspaceAllMountainDew:GetChildren()) do
		local isBurned = d:FindFirstChild("IsBurned")
		if (isBurned == nil or isBurned.Value == false) and d.Position.X > 53 and d.Position.Z > 50 and not d.Anchored then
			return d
		end
	end
end

local function FindDoughAndWithout(str)
	if not workspaceAllDough then return end
	local goodraw, p, raw, trash
	local children = workspaceAllDough:GetChildren()
	
	for i = #children, 2, -1 do
		local j = RNG:NextInteger(1, i)
		children[j], children[i] = children[i], children[j]
	end
	
	for _, d in ipairs(children) do
		if d.Anchored == false and #d:GetChildren() > 9 then
			local isBurned = d:FindFirstChild("IsBurned")
			local hasBugs = d:FindFirstChild("HasBugs")
			local cold = d:FindFirstChild("Cold")
			local sg = d:FindFirstChild("SG")
			
			if isBurned and hasBuses and cold then
				if isBurned.Value or hasBugs.Value or cold.Value then
					if trash == nil and d.Position.Y > 0 then
						trash = d
					end
				elseif p == nil and d.BrickColor.Name == "Bright orange" then
					p = d
				elseif goodraw == nil and d.Position.X < 55 and d.BrickColor.Name == "Brick yellow" then
					if sg and sg:FindFirstChild("Frame") then
						local checkPassed = str and not sg.Frame:FindFirstChild(str) or (sg.Frame:FindFirstChild("Sausage") == nil and sg.Frame:FindFirstChild("Pepperoni") == nil)
						if checkPassed then
							local mesh = d:FindFirstChild("Mesh")
							if mesh then
								if mesh.Scale.Y < 1.1 then
									goodraw = d
								else
									raw = d
								end
							end
						end
					end
				end
			end
		end
	end
	return goodraw or raw, p, trash
end

local function getOvenNear(pos)
	if not workspaceOvens then return end
	for _, o in ipairs(workspaceOvens:GetChildren()) do
		local bottom = o:FindFirstChild("Bottom")
		if bottom and (bottom.Position - pos).magnitude < 1.5 then
			return o
		end
	end
end

local function isFullyOpen(oven)
	local isOpen = oven:FindFirstChild("IsOpen")
	if not isOpen or not isOpen.Value then return false end
	local door = oven:FindFirstChild("Door")
	if not door then return false end
	local meter = door:FindFirstChild("Meter")
	if not meter then return false end
	local rotVel = meter:FindFirstChild("RotVelocity")
	if not rotVel then return false end
	return (rotVel.Z ^ 2) ^ .5 < .0001
end

local bcolorToSupply = {["Dark orange"] = "Sausage", ["Bright blue"] = "Pepperoni", ["Bright yellow"] = "Cheese", ["Bright red"] = "TomatoSauce", ["Dark green"] = "Dew", ["Brick yellow"] = "Dough", ["Light stone grey"] = "Box", ["Really black"] = "Dew"}
local supplyButtons = {}
if workspaceSupplyButtons then
	for _, btnContainer in ipairs(workspaceSupplyButtons:GetChildren()) do
		local unpressed = btnContainer:FindFirstChild("Unpressed")
		if unpressed then
			supplyButtons[#supplyButtons + 1] = unpressed
		end
	end
end

-- Main Loop
task.spawn(function()
	while gui.Parent do
		task.wait(0.4)
		if humanoid then humanoid.Sit = false end
		
		-- Cooking loop block
		if doCook and workspaceOvens and network then
			pcall(function()
				local orders = getOrders()
				if #orders > 0 then
					local order = orders[1]
					local cookD = order == "Dew" and FindFirstDew() or nil
					if order == "Dew" and cookD then
						network:FireServer("UpdateProperty", cookD, "CFrame", CFrame.new(53, 4.68, 36.5))
					end
				end
			end)
		end
		
		-- Supplying loop block
		if doSupplier and workspaceAllSupplyBoxes and network then
			pcall(function()
				local refill = false
				for s, c in pairs(supplyCounts) do
					if c <= 20 then
						refill = true
						break
					end
				end
				if refill and #supplyButtons > 0 then
					for _, btn in ipairs(supplyButtons) do
						local s = bcolorToSupply[btn.BrickColor.Name]
						if s and supplyCounts[s] < 60 then
							instantTP(btn.CFrame + Vector3.new(0, 3, 0))
							task.wait(0.1)
						end
					end
				end
			end)
		end
	end
end)
